
#VirSorter2
virsorter run -w VIRSorter_predicted_viral_fna -i pooled.contigs.l500.fasta --min-length 500 -j 8 all

#BLASTN search against Viral Database
makeblastdb -in viral.1.1.genomic.fna -dbtype nucl -out viral_db

blastn -query VIRSorter_predicted_viral_fna/viral-combined-cats.fasta -db viral_db -out blastn_results.tsv -evalue 0.005 -perc_identity 70 -outfmt '6 qseqid sseqid salltitles pident length mismatch gapopen qstart qend sstart send evalue bitscore' -max_target_seqs 1

#Reads Abundance Calculation
bwa index VIRSorter_predicted_viral_fna/viral-combined-cats.fasta

bwa mem VIRSorter_predicted_viral_fna/viral-combined-cats.fasta pooled-all.fastq | samtools view -b | samtools sort -T viral > viral.sort.bam
samtools index viral.sort.bam
samtools idxstats viral.sort.bam > viral.idxstats.tsv

#Gene Prediction
prodigal -a viral.proteins.fasta -c -d $output_dir/viral.nucleotides.fasta -f gff -i viral-combined-cats.fasta -o output_dir/viral.gff


#Viral Binning
vRhyme -i viral-combined-cats.fasta -b viral.sort.bam
